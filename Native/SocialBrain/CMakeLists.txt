cmake_minimum_required(VERSION 3.24)
project(FratRowSocialNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(RUNANYWHERE_ROOT "" CACHE PATH "Verified cpp-desktop-v0.20.39 prefix")
if(NOT WIN32 OR NOT EXISTS "${RUNANYWHERE_ROOT}/lib/rac_backend_llamacpp.lib")
  message(FATAL_ERROR "Windows x64 and the pinned RunAnywhere desktop kit are required")
endif()
file(READ "${RUNANYWHERE_ROOT}/share/runanywhere/VERSION" SDK_VERSION)
string(STRIP "${SDK_VERSION}" SDK_VERSION)
if(NOT SDK_VERSION STREQUAL "0.20.39")
  message(FATAL_ERROR "Expected RunAnywhere 0.20.39, got ${SDK_VERSION}")
endif()
add_library(FratRowSocialNative SHARED social_brain.cpp)
target_include_directories(FratRowSocialNative PRIVATE "${RUNANYWHERE_ROOT}/include")
# Intentionally avoid the kit's umbrella target: it whole-archives speech backends.
# Only text inference is referenced. No desktop adapter or HTTP transport is registered.
file(GLOB SUPPORT_LIBS "${RUNANYWHERE_ROOT}/lib/absl_*.lib")
foreach(NAME rac_backend_llamacpp rac_commons llama llama-common llama-common-base ggml ggml-base ggml-cpu rac_protobuf_isolated libutf8_validity archive bz2_bundled zlibstatic)
  list(APPEND TEXT_LIBS "${RUNANYWHERE_ROOT}/lib/${NAME}.lib")
endforeach()
target_link_libraries(FratRowSocialNative PRIVATE ${TEXT_LIBS} ${SUPPORT_LIBS} ws2_32 bcrypt crypt32 advapi32 iphlpapi xmllite ole32)
target_compile_options(FratRowSocialNative PRIVATE /EHsc /utf-8)
